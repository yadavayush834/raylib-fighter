cmake_minimum_required(VERSION 3.20)
project(spirit_clash VERSION 0.2.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(SPIRIT_WAYLAND_DEFAULT OFF)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND DEFINED ENV{WAYLAND_DISPLAY})
    set(SPIRIT_WAYLAND_DEFAULT ON)
endif()
option(SPIRIT_NATIVE_WAYLAND "Build a native Wayland raylib backend on Linux" ${SPIRIT_WAYLAND_DEFAULT})
if(SPIRIT_NATIVE_WAYLAND)
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WAYLAND_DEPS REQUIRED wayland-client wayland-cursor wayland-egl xkbcommon)
    set(GLFW_BUILD_WAYLAND ON CACHE BOOL "" FORCE)
    set(GLFW_BUILD_X11 OFF CACHE BOOL "" FORCE)
else()
    find_package(raylib 6.0 QUIET)
endif()
if(SPIRIT_NATIVE_WAYLAND OR NOT raylib_FOUND)
    include(FetchContent)
    set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
    set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 6.2 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(raylib)
endif()
add_library(combat STATIC src/combat.c)
target_include_directories(combat PUBLIC src)
target_link_libraries(combat PUBLIC raylib)
if(UNIX)
    target_link_libraries(combat PUBLIC m)
endif()
add_executable(spirit-clash main.c src/render.c src/audio.c)
target_link_libraries(spirit-clash PRIVATE combat)
target_compile_definitions(spirit-clash PRIVATE ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
if(MSVC)
    target_compile_options(combat PRIVATE /W4)
    target_compile_options(spirit-clash PRIVATE /W4)
else()
    target_compile_options(combat PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(spirit-clash PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
    add_executable(combat-tests tests/combat_test.c)
    target_link_libraries(combat-tests PRIVATE combat)
    add_test(NAME combat-regressions COMMAND combat-tests)
endif()
